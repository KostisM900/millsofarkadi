Κανένα ποσό. -- Χωρίς σύνδεση (email + κωδικό) δεν διαβάζεται και δεν γράφεται τίποτα. -- ===================================================================== -- Φάκελοι στο stand create table if not exists public.envelopes ( id uuid primary key default gen_random_uuid(), name text not null check (char_length(name) between 1 and 80), date date, -- ημέρα παράδοσης (μπορεί να είναι κενή) kind text check (kind in ('apodeixi','timologio')), -- είδος: απόδειξη ή τιμολόγιο note text check (char_length(note) <= 200), -- σχόλιο (προαιρετικό, χωρίς ποσά) added_at timestamptz not null default now(), -- πότε μπήκε στο stand given_at timestamptz, -- πότε δόθηκε (κενό = ακόμα στο stand) created_by uuid default auth.uid() ); -- Πελάτες που θυμάται η εφαρμογή (για τις προτάσεις ονομάτων) create table if not exists public.customers ( id uuid primary key default gen_random_uuid(), name text not null check (char_length(name) between 1 and 80), name_key text not null unique, -- όνομα χωρίς τόνους/κεφαλαία, για να μη γράφεται δύο φορές created_at timestamptz not null default now() ); -- Για βάση που φτιάχτηκε με την προηγούμενη έκδοση: προσθέτει είδος και σχόλιο. -- (Ασφαλές να τρέξει ξανά, δεν σβήνει τίποτα.) alter table public.envelopes add column if not exists kind text check (kind in ('apodeixi','timologio')); alter table public.envelopes add column if not exists note text check (char_length(note) <= 200); create index if not exists envelopes_pending_idx on public.envelopes (date) where given_at is null; -- Ασφάλεια: κλειδώνουμε τους πίνακες και ανοίγουμε μόνο σε συνδεδεμένους χρήστες alter table public.envelopes enable row level security; alter table public.customers enable row level security; drop policy if exists "envelopes_authenticated" on public.envelopes; create policy "envelopes_authenticated" on public.envelopes for all to authenticated using (true) with check (true); drop policy if exists "customers_authenticated" on public.customers; create policy "customers_authenticated" on public.customers for all to authenticated using (true) with check (true); -- Ο ανώνυμος ρόλος (όποιος έχει μόνο το δημόσιο κλειδί) δεν έχει καμία πρόσβαση revoke all on public.envelopes from anon; revoke all on public.customers from anon; -- ===================================================================== -- Έλεγχος αρχειοθέτησης παραστατικών · ΜΟΝΟ για τον λογαριασμό social@m-m-m.gr -- Αποθηκεύονται μόνο: αριθμός παραστατικού, ημερομηνία του, αν αρχειοθετήθηκε. -- Οι εικόνες δεν ανεβαίνουν ποτέ (η αναγνώριση γίνεται στον browser). -- ===================================================================== create table if not exists public.archive_docs ( code text primary key check (char_length(code) between 3 and 30), doc_date date, archived_at timestamptz, -- κενό = λείπει από την αρχειοθέτηση created_at timestamptz not null default now() ); create index if not exists archive_docs_missing_idx on public.archive_docs (code) where archived_at is null; create table if not exists public.archive_checks ( id uuid primary key default gen_random_uuid(), label text check (char_length(label) <= 120), codes text[] not null default '{}', -- τα παραστατικά του κάθε στιγμιότυπου created_at timestamptz not null default now() ); alter table public.archive_docs enable row level security; alter table public.archive_checks enable row level security; drop policy if exists "archive_docs_social" on public.archive_docs; create policy "archive_docs_social" on public.archive_docs for all to authenticated using (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr'); drop policy if exists "archive_checks_social" on public.archive_checks; create policy "archive_checks_social" on public.archive_checks for all to authenticated using (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr'); revoke all on public.archive_docs from anon; revoke all on public.archive_checks from anon; -- ===================================================================== -- Ζωντανή ενημέρωση (Realtime): κάθε αλλαγή φτάνει αμέσως στους άλλους υπολογιστές -- Χρειάζεται μία φορά· ασφαλές να τρέξει ξανά. -- ===================================================================== do $$ declare t text; begin foreach t in array array['envelopes','customers','archive_docs','archive_checks'] loop begin execute format('alter publication supabase_realtime add table public.%I', t); exception when duplicate_object then null; -- ήδη προστέθηκε end; end loop; end $$; -- Οι ειδοποιήσεις ακολουθούν τους ίδιους κανόνες RLS: ο καθένας βλέπει μόνο ό,τι δικαιούται. alter table public.envelopes replica identity full; alter table public.customers replica identity full; alter table public.archive_docs replica identity full; alter table public.archive_checks replica identity full; -- ===================================================================== -- Επιταγές: υπενθύμιση για την ημερομηνία εξόφλησης -- ΜΟΝΟ για τον λογαριασμό social@m-m-m.gr. -- Αποθηκεύονται ΜΟΝΟ: ημερομηνία, τα 4 τελευταία ψηφία (προαιρετικά), -- σύντομη σημείωση και η κατάσταση. -- ΠΟΤΕ: αριθμός επιταγής, τράπεζα, λογαριασμός, εκδότης, ποσό, πελάτης. -- ===================================================================== create table if not exists public.cheques ( id uuid primary key default gen_random_uuid(), due_date date not null, -- πότε εξοφλείται last4 text check (last4 ~ '^[0-9]{4}$'), -- μόνο 4 ψηφία, τίποτα άλλο note text check (char_length(note) <= 80), status text not null default 'pending' check (status in ('pending','cleared')), settled_at timestamptz, created_at timestamptz not null default now() ); -- για βάση που φτιάχτηκε με την προηγούμενη έκδοση do $$ begin execute 'alter table public.cheques alter column customer drop not null'; exception when undefined_column then null; end $$; alter table public.cheques drop column if exists customer; create index if not exists cheques_pending_idx on public.cheques (due_date) where status = 'pending'; alter table public.cheques enable row level security; drop policy if exists "cheques_authenticated" on public.cheques; drop policy if exists "cheques_social" on public.cheques; create policy "cheques_social" on public.cheques for all to authenticated using (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check (lower(auth.jwt() ->> 'email') = 'social@m-m-m.gr'); revoke all on public.cheques from anon; do $$ begin begin execute 'alter publication supabase_realtime add table public.cheques'; exception when duplicate_object then null; end; end $$; alter table public.cheques replica identity full; -- ===================================================================== -- ΕΛΕΓΧΟΣ ΑΣΦΑΛΕΙΑΣ — τρέξε το και δες το αποτέλεσμα -- Όλες οι γραμμές πρέπει να λένε "ΟΚ". Αν κάποια λέει ΠΡΟΣΟΧΗ, κάτι είναι ανοιχτό. -- ===================================================================== -- 1) Ο ανώνυμος ρόλος (αυτός που έχει μόνο το δημόσιο κλειδί) να μην αγγίζει τίποτα revoke all on all tables in schema public from anon; revoke all on all sequences in schema public from anon; revoke all on all functions in schema public from anon; revoke usage on schema public from anon; alter default privileges in schema public revoke all on tables from anon; -- 2) Έλεγχος: RLS ενεργό παντού select tablename as "Πίνακας", case when rowsecurity then 'ΟΚ — RLS ενεργό' else 'ΠΡΟΣΟΧΗ — RLS κλειστό!' end as "Κατάσταση", (select count(*) from pg_policies p where p.tablename = t.tablename and p.schemaname = 'public') as "Κανόνες" from pg_tables t where schemaname = 'public' order by tablename; -- 3) Έλεγχος: τι μπορεί να κάνει ο ανώνυμος ρόλος (πρέπει να μη γυρίσει καμία γραμμή) select table_name as "Πίνακας", privilege_type as "Δικαίωμα του anon — ΠΡΟΣΟΧΗ" from information_schema.role_table_grants where grantee = 'anon' and table_schema = 'public';
