Χρώματα ---------- -- Το hex/lab κρατιέται για να αναγνωρίζει η κάμερα το χρώμα. create table if not exists public.stock_colors ( name text primary key, -- ΚΙΤΡΙΝΟ, ΜΠΕΖ, ΓΚΡΙ, ΜΠΛΕ… hex text not null default '#CCCCCC', lab_l double precision, lab_a double precision, lab_b double precision, created_at timestamptz not null default now() ); -- ---------- 2. Παραλλαγές: είδος + χρώμα + μέγεθος ---------- create table if not exists public.stock_variants ( id uuid primary key default gen_random_uuid(), kind text not null default 'short', -- short = κοντομάνικο, long = μακρυμάνικο color text not null, size text not null, barcode text, -- προαιρετικό, αν κάποτε σκανάρεις σακουλάκια qty integer not null default 0, min_qty integer not null default 0, created_at timestamptz not null default now() ); create unique index if not exists stock_variants_combo_key on public.stock_variants (kind, color, size); create unique index if not exists stock_variants_barcode_key on public.stock_variants (barcode) where barcode is not null; -- ---------- 3. Κινήσεις: το ιστορικό κάθε αλλαγής ---------- create table if not exists public.stock_moves ( id uuid primary key default gen_random_uuid(), variant_id uuid not null references public.stock_variants(id) on delete cascade, delta integer not null, -- +3 μπήκαν, -2 βγήκαν kind text not null default 'in', -- in | out | count | cam note text not null default '', by_email text not null default '', at timestamptz not null default now() ); create index if not exists stock_moves_at_idx on public.stock_moves (at desc); -- ---------- 4. Αλλαγή ποσότητας + καταγραφή κίνησης, με μία κλήση ---------- create or replace function public.stock_bump( p_variant uuid, p_delta integer, p_kind text default 'in', p_note text default '' ) returns integer language plpgsql security invoker as $$ declare v_qty integer; begin update public.stock_variants set qty = greatest(0, qty + p_delta) where id = p_variant returning qty into v_qty; if v_qty is null then raise exception 'unknown variant %', p_variant; end if; insert into public.stock_moves (variant_id, delta, kind, note, by_email) values (p_variant, p_delta, coalesce(p_kind, 'in'), coalesce(p_note, ''), coalesce(auth.jwt() ->> 'email', '')); return v_qty; end; $$; -- ---------- 5. Απογραφή: απόλυτο νούμερο, κρατάει τη διαφορά ---------- create or replace function public.stock_set( p_variant uuid, p_qty integer, p_note text default '' ) returns integer language plpgsql security invoker as $$ declare v_old integer; v_new integer := greatest(0, p_qty); begin select qty into v_old from public.stock_variants where id = p_variant; if v_old is null then raise exception 'unknown variant %', p_variant; end if; update public.stock_variants set qty = v_new where id = p_variant; if v_new <> v_old then insert into public.stock_moves (variant_id, delta, kind, note, by_email) values (p_variant, v_new - v_old, 'count', coalesce(p_note, ''), coalesce(auth.jwt() ->> 'email', '')); end if; return v_new; end; $$; -- ---------- 6. Κλείδωμα: μόνο ο social@m-m-m.gr ---------- alter table public.stock_colors enable row level security; alter table public.stock_variants enable row level security; alter table public.stock_moves enable row level security; drop policy if exists stock_colors_social on public.stock_colors; create policy stock_colors_social on public.stock_colors for all to authenticated using ((auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check ((auth.jwt() ->> 'email') = 'social@m-m-m.gr'); drop policy if exists stock_variants_social on public.stock_variants; create policy stock_variants_social on public.stock_variants for all to authenticated using ((auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check ((auth.jwt() ->> 'email') = 'social@m-m-m.gr'); drop policy if exists stock_moves_social on public.stock_moves; create policy stock_moves_social on public.stock_moves for all to authenticated using ((auth.jwt() ->> 'email') = 'social@m-m-m.gr') with check ((auth.jwt() ->> 'email') = 'social@m-m-m.gr'); -- Οι συναρτήσεις τρέχουν με τα δικαιώματα του χρήστη (security invoker), -- άρα τα παραπάνω RLS ισχύουν και μέσα από αυτές. revoke all on function public.stock_bump(uuid, integer, text, text) from public, anon; revoke all on function public.stock_set(uuid, integer, text) from public, anon; grant execute on function public.stock_bump(uuid, integer, text, text) to authenticated; grant execute on function public.stock_set(uuid, integer, text) to authenticated;
